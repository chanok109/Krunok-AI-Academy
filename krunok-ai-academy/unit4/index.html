AApply: สร้างต้นแบบจากแม่แบบโค้ด

ใช้แม่แบบที่รวมกล้อง โมเดล กติกา การแสดงผล และการบันทึกผลทดสอบไว้แล้ว กลุ่มแก้แค่ 3 ส่วน ใช้ได้ทั้งโมเดลจำแนกภาพจากหัวข้อ 3.2 และโมเดลตรวจจับวัตถุจากหัวข้อ 3.3

MODEL, CLASSES1) ตั้งค่าที่อยู่ไฟล์โมเดลของกลุ่ม และชื่อประเภทตามลำดับ
def decide(label, conf)2) กติกาเปลี่ยนคำตอบของโมเดลเป็นการกระทำ ส่วนที่ทำให้ชิ้นงานต่างจากกลุ่มอื่น
test_log.csv3) บันทึกผลตอนทดสอบกดปุ่มบอกคำตอบที่ถูก ระบบบันทึกคู่กับคำตอบของโมเดล

ทำไมต้องมี "ไม่แน่ใจ": โมเดลตอบได้แค่ประเภทที่เรียนมา และตอบอย่างมั่นใจแม้เจอสิ่งที่ไม่รู้จัก (เห็นแล้วในหัวข้อ 3.2) กติกาที่ดีจึงควรมีทางเลือกเมื่อความมั่นใจต่ำ เช่น ขอให้ถือวัตถุใหม่ หรือให้คนตัดสิน

ลงมือเขียนโค้ดจริง

  1. เตรียมโฟลเดอร์สร้างโฟลเดอร์ D:\cv\unit4 คัดลอกโฟลเดอร์ runs ที่มีโมเดลของกลุ่มมาไว้ด้วย
  2. คัดลอกแม่แบบสร้างไฟล์ 01_project_template.py แล้วแก้ MODEL CLASSES และฟังก์ชัน decide
    # แม่แบบโครงงาน AI: กล้อง → โมเดล → ตัดสินใจ → แสดงผล → บันทึก
import cv2
import csv
import time
from ultralytics import YOLO

# ===== 1) ตั้งค่าโครงงาน (แก้ส่วนนี้) =====
MODEL = "runs/classify/emoji/weights/best.pt"   # โมเดลของกลุ่ม หรือ "yolo11n.pt" สำหรับตรวจจับวัตถุ
CLASSES = ["happy", "sad", "surprised"]         # ประเภทที่โมเดลรู้จัก (กดปุ่ม 1, 2, 3 = คำตอบที่ถูกตอนทดสอบ)
MIN_CONF = 0.7                                   # มั่นใจน้อยกว่านี้ ถือว่า "ไม่แน่ใจ"
SOURCE = 0                                       # 0 = กล้อง หรือใส่ชื่อไฟล์วิดีโอ

def decide(label, conf):
    # ===== 2) กติกาของโครงงาน: เปลี่ยนคำตอบของโมเดลเป็นการกระทำ =====
    if conf < MIN_CONF:
        return "uncertain - show again"
    if label == "happy":
        return "Great! keep smiling"
    if label == "sad":
        return "Take a break?"
    return "Wow!"

model = YOLO(MODEL)
cap = cv2.VideoCapture(SOURCE)
if not cap.isOpened():
    print("เปิดกล้องหรือวิดีโอไม่ได้")
    raise SystemExit

log = open("test_log.csv", "a", newline="")       # ===== 3) บันทึกผลทดสอบ =====
writer = csv.writer(log)
tested = 0
print("กด 1 ถึง", len(CLASSES), "บอกคำตอบที่ถูกเพื่อบันทึกผลทดสอบ กด q เพื่อออก")

while True:
    ok, frame = cap.read()
    if not ok:
        break
    t = time.time()
    r = model(frame, verbose=False)[0]
    if r.probs is not None:                       # โมเดลจำแนกภาพ
        label, conf = r.names[r.probs.top1], float(r.probs.top1conf)
    elif len(r.boxes) > 0:                        # โมเดลตรวจจับวัตถุ ใช้กรอบที่มั่นใจที่สุด
        best = r.boxes[int(r.boxes.conf.argmax())]
        label, conf = r.names[int(best.cls)], float(best.conf)
    else:
        label, conf = "none", 0.0
    action = decide(label, conf)
    fps = 1 / max(time.time() - t, 1e-6)

    cv2.putText(frame, f"{label} {conf:.2f}", (10, 35), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 200, 0), 2)
    cv2.putText(frame, action, (10, 75), cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 0, 255), 2)
    cv2.putText(frame, f"FPS {fps:.1f}  tested {tested}", (10, frame.shape[0] - 15),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2)
    cv2.imshow("AI Project", frame)

    key = cv2.waitKey(1) & 0xFF
    if key == ord("q"):
        break
    for i, c in enumerate(CLASSES):
        if key == ord(str(i + 1)):                # กดคำตอบที่ถูก → บันทึกคู่ (คำตอบจริง, โมเดลทาย)
            writer.writerow([c, label, round(conf, 3)])
            tested = tested + 1

log.close()
cap.release()
cv2.destroyAllWindows()
print("บันทึกผลทดสอบเพิ่ม", tested, "ครั้ง ลงไฟล์ test_log.csv")
  3. รันต้นแบบถือวัตถุหน้ากล้อง ดูคำตอบและการกระทำ กดปุ่ม q เพื่อปิด

สิ่งที่จะเห็น

บรรทัดแรกคือคำตอบของโมเดลกับความมั่นใจ บรรทัดที่สองคือการกระทำจาก decide ด้านล่างคือ FPS และจำนวนครั้งที่ทดสอบแล้ว ภาพตัวอย่างได้จากการรันแม่แบบจริงกับโมเดลอีโมจิจากหัวข้อ 3.2 โดยป้อนวิดีโอแทนกล้อง

ลงมือเขียนโค้ดจริง: วัดผลจากบันทึกการทดสอบ

  1. ทดสอบด้วยแม่แบบรัน 01_project_template.py ทุกครั้งที่ภาพนิ่ง ให้กดปุ่มตัวเลขของคำตอบที่ถูก ประเภทละอย่างน้อย 10 ครั้ง ในหลายสภาพ
  2. คัดลอกโค้ดวัดผลสร้างไฟล์ 02_evaluate.py แก้ CLASSES ให้ตรงกับแม่แบบ
    import csv
from sklearn.metrics import accuracy_score, confusion_matrix, classification_report

CLASSES = ["happy", "sad", "surprised"]
y_true, y_pred = [], []
with open("test_log.csv") as f:
    for true, pred, conf in csv.reader(f):
        y_true.append(true)
        y_pred.append(pred)

print("จำนวนครั้งที่ทดสอบ:", len(y_true))
print("ความแม่นยำ:", round(accuracy_score(y_true, y_pred), 3))
print("ตารางความสับสน", CLASSES)
print(confusion_matrix(y_true, y_pred, labels=CLASSES))
print(classification_report(y_true, y_pred, labels=CLASSES, zero_division=0))

  • สีหน้าไม่ได้บอกความตั้งใจได้แม่นยำ บางคนตั้งใจฟังแต่หน้านิ่ง ถ้าระบบทายผิด นักเรียนอาจถูกตำหนิโดยไม่เป็นธรรม
  • เป็นการเก็บภาพใบหน้าทุกคาบโดยนักเรียนไม่ได้เลือก กระทบความเป็นส่วนตัว และสร้างความกดดัน
  • ปรับได้โดยเปลี่ยนคำถาม เช่น ให้นักเรียนชูมือเป็นท่าบอกว่าเข้าใจหรือยัง (หัวข้อ 3.5) แล้วรายงานเป็นภาพรวมทั้งห้อง ไม่ระบุตัวคน
